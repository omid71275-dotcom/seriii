  <script>
    // ========== Telegram WebApp ==========
    const tg = window.Telegram.WebApp;
    tg.ready();
    tg.expand();
    tg.setHeaderColor('#0f0f1a');
    tg.setBackgroundColor('#0f0f1a');

    // ========== Canvas Setup ==========
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');

    let width, height;
    function resize() {
      const container = document.getElementById('game-container');
      width = container.clientWidth;
      height = container.clientHeight;
      canvas.width = width;
      canvas.height = height;
    }
    resize();
    window.addEventListener('resize', resize);

    // ========== Game Variables ==========
    const paddleWidth = 80;
    const paddleHeight = 12;
    const ballSize = 10;

    let player = { x: 0, y: 0, width: paddleWidth, height: paddleHeight, score: 0 };
    let ai = { x: 0, y: 0, width: paddleWidth, height: paddleHeight, score: 0 };
    let ball = { x: 0, y: 0, vx: 0, vy: 0, size: ballSize };

    let isPlaying = false;
    let animationId = null;

    // ========== Init Positions ==========
    function resetPositions() {
      player.x = width / 2 - paddleWidth / 2;
      player.y = height - 40;

      ai.x = width / 2 - paddleWidth / 2;
      ai.y = 28;

      ball.x = width / 2;
      ball.y = height / 2;

      // Direction random
      const angle = (Math.random() * Math.PI / 3) - Math.PI / 6; // -30 to 30 deg
      const speed = 5;
      const direction = Math.random() > 0.5 ? 1 : -1;
      ball.vx = Math.sin(angle) * speed * direction;
      ball.vy = Math.cos(angle) * speed * (Math.random() > 0.5 ? 1 : -1);
    }

    // ========== Controls ==========
    function movePaddle(clientX) {
      const rect = canvas.getBoundingClientRect();
      const x = clientX - rect.left;
      player.x = Math.max(0, Math.min(width - paddleWidth, x - paddleWidth / 2));
    }

    canvas.addEventListener('mousemove', e => {
      if (isPlaying) movePaddle(e.clientX);
    });

    canvas.addEventListener('touchmove', e => {
      e.preventDefault();
      if (isPlaying) movePaddle(e.touches[0].clientX);
    }, { passive: false });

    canvas.addEventListener('touchstart', e => {
      e.preventDefault();
      if (isPlaying) movePaddle(e.touches[0].clientX);
    }, { passive: false });

    // ========== AI ==========
    function updateAI() {
      const aiCenter = ai.x + paddleWidth / 2;
      const target = ball.x;
      const diff = target - aiCenter;
      const speed = 4.2; // AI difficulty

      if (Math.abs(diff) > 6) {
        ai.x += Math.sign(diff) * speed;
        ai.x = Math.max(0, Math.min(width - paddleWidth, ai.x));
      }
    }

    // ========== Collision ==========
    function checkCollision(paddle) {
      return (
        ball.x + ball.size > paddle.x &&
        ball.x - ball.size < paddle.x + paddle.width &&
        ball.y + ball.size > paddle.y &&
        ball.y - ball.size < paddle.y + paddle.height
      );
    }

    // ========== Update ==========
    function update() {
      if (!isPlaying) return;

      // Move ball
      ball.x += ball.vx;
      ball.y += ball.vy;

      // Walls left/right
      if (ball.x - ball.size < 0 || ball.x + ball.size > width) {
        ball.vx *= -1;
      }

      // Player paddle
      if (checkCollision(player) && ball.vy > 0) {
        ball.vy *= -1;
        // Add some spin based on hit position
        const hitPos = (ball.x - (player.x + paddleWidth / 2)) / (paddleWidth / 2);
        ball.vx = hitPos * 5;
        // Slightly increase speed
        ball.vy *= 1.03;
      }

      // AI paddle
      if (checkCollision(ai) && ball.vy < 0) {
        ball.vy *= -1;
        const hitPos = (ball.x - (ai.x + paddleWidth / 2)) / (paddleWidth / 2);
        ball.vx = hitPos * 5;
        ball.vy *= 1.03;
      }

      // Score
      if (ball.y > height) {
        ai.score++;
        updateScore();
        if (ai.score >= 5) endGame(false);
        else resetBall();
      }

      if (ball.y < 0) {
        player.score++;
        updateScore();
        if (player.score >= 5) endGame(true);
        else resetBall();
      }

      updateAI();
    }

    function resetBall() {
      ball.x = width / 2;
      ball.y = height / 2;
      const angle = (Math.random() * Math.PI / 3) - Math.PI / 6;
      const speed = 5;
      ball.vx = Math.sin(angle) * speed * (Math.random() > 0.5 ? 1 : -1);
      ball.vy = Math.cos(angle) * speed * (Math.random() > 0.5 ? 1 : -1);
    }

    function updateScore() {
      document.getElementById('score').textContent = `${player.score} : ${ai.score}`;
    }

    // ========== Draw ==========
    function draw() {
      // Clear
      ctx.fillStyle = '#0f0f1a';
      ctx.fillRect(0, 0, width, height);

      // Center line
      ctx.strokeStyle = 'rgba(255,255,255,0.15)';
      ctx.setLineDash([8, 12]);
      ctx.beginPath();
      ctx.moveTo(0, height / 2);
      ctx.lineTo(width, height / 2);
      ctx.stroke();
      ctx.setLineDash([]);

      // Paddles
      ctx.fillStyle = '#00ff9d';
      // Rounded paddles
      roundRect(player.x, player.y, player.width, player.height, 6);
      roundRect(ai.x, ai.y, ai.width, ai.height, 6);

      // Ball
      ctx.beginPath();
      ctx.arc(ball.x, ball.y, ball.size, 0, Math.PI * 2);
      ctx.fillStyle = '#ffffff';
      ctx.fill();
      ctx.shadowBlur = 15;
      ctx.shadowColor = '#00ff9d';
      ctx.fill();
      ctx.shadowBlur = 0;
    }

    function roundRect(x, y, w, h, r) {
      ctx.beginPath();
      ctx.moveTo(x + r, y);
      ctx.arcTo(x + w, y, x + w, y + h, r);
      ctx.arcTo(x + w, y + h, x, y + h, r);
      ctx.arcTo(x, y + h, x, y, r);
      ctx.arcTo(x, y, x + w, y, r);
      ctx.closePath();
      ctx.fill();
    }

    // ========== Game Loop ==========
    function gameLoop() {
      update();
      draw();
      animationId = requestAnimationFrame(gameLoop);
    }

    // ========== Start / End ==========
    function startGame() {
      player.score = 0;
      ai.score = 0;
      updateScore();
      resetPositions();
      isPlaying = true;

      document.getElementById('start-screen').classList.add('hidden');
      document.getElementById('game-over').classList.add('hidden');

      if (animationId) cancelAnimationFrame(animationId);
      gameLoop();
    }

    function endGame(playerWon) {
      isPlaying = false;
      cancelAnimationFrame(animationId);

      const winnerText = document.getElementById('winnerText');
      const finalScore = document.getElementById('finalScore');

      if (playerWon) {
        winnerText.textContent = 'برنده شدی! 🎉';
        winnerText.style.color = '#00ff9d';
      } else {
        winnerText.textContent = 'باختی 😅';
        winnerText.style.color = '#ff6b6b';
      }

      finalScore.textContent = `${player.score} : ${ai.score}`;
      document.getElementById('game-over').classList.remove('hidden');
    }

    // ========== Buttons ==========
    document.getElementById('startBtn').addEventListener('click', startGame);
    document.getElementById('restartBtn').addEventListener('click', startGame);

    // Initial draw
    resetPositions();
    draw();
  </script>
